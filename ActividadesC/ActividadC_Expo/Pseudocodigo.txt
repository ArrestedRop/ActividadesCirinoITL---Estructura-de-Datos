        ESCRIBE("2. Eliminar objetos del stack\n")
        ESCRIBE("3. Ver el tope del stack\n")
        ESCRIBE("4. Verificar si el stack esta vacio\n")
        ESCRIBE("5. Verificar si el stack esta lleno\n")
        ESCRIBE("6. Salir\n")

        RECIBIR choice
        SI (choice != 1) HACER:
            ESCRIBIR("\n[!] ERROR: Entrada invalida. Por favor, ingresa un numero.\n")
            
                MIENTRAS ((c = getchar()) != '\n' && c != EOF);
                choice = 0; 
                CONTINUAR;
        TERMINAR
        SI(choice < 1 || choice > 6) HACER:
            ESCRIBIR("\n[!] ERROR: Opcion fuera de rango. Elige un numero del 1 al 6.\n")
        O SI(choice != 6): ESCRIBISTE("\nElegiste la opcion %d...\n", choice)
    MIENTRAS(choice != 0) TERMINAR
    REGRESAR choice
TERMINAR METODO
